  minCyc     = admitted FIFO-B entries; 1 activation/cycle with no stalls
  cmpCyc     = streaming cycles = minCyc + fetch-stall beats
  latency    = end-to-end: cmpCyc + weight-load + arm + drain
  speedup    = roofline cmpCyc / this case's cmpCyc

  macs/act   = useful MACs per admitted activation (broadcast efficiency)
